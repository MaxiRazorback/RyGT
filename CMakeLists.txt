function(add_test_executable target)
    # The runtime library test sources are not published; keep their targets (other CMake code links
    # and configures them) but build them from a stub, outside the default build, when sources are missing.
    set(sources "")
    set(target_options "")
    foreach(source ${ARGN})
        if(source STREQUAL "EXCLUDE_FROM_ALL")
            list(APPEND target_options EXCLUDE_FROM_ALL)
        elseif(EXISTS "${source}" OR EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
            list(APPEND sources "${source}")
        endif()
    endforeach()
    if(sources)
        add_executable(${target} ${target_options} ${sources})
    else()
        set(stub "${CMAKE_BINARY_DIR}/missing_test_stub.cpp")
        if(NOT EXISTS "${stub}")
            file(WRITE "${stub}" "int main() { return 0; }\n")
        endif()
        add_executable(${target} EXCLUDE_FROM_ALL "${stub}")
    endif()
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${RTY_TEST_OUTPUT_DIRECTORY}>" INTERPROCEDURAL_OPTIMIZATION FALSE)
endfunction()

# The runtime libraries, shader recompiler and media decoders depend on the
# 3rdparty submodules. When they are missing (fresh clone without
# `git submodule update --init`), only the relinker CLI and its tests build.
set(RTY_HAVE_THIRD_PARTY ON)
foreach(module SDL2/CMakeLists.txt ffmpeg-core/CMakeLists.txt freetype/CMakeLists.txt stb/stb_image.h)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/3rdparty/${module}")
        set(RTY_HAVE_THIRD_PARTY OFF)
    endif()
endforeach()

if(RTY_HAVE_THIRD_PARTY)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    set(SDL_VIDEO ON CACHE BOOL "" FORCE)
    set(SDL_RENDER OFF CACHE BOOL "" FORCE)
    set(SDL_X11 ON CACHE BOOL "" FORCE)
    set(SDL_WAYLAND OFF CACHE BOOL "" FORCE)
    set(SDL_OPENGL OFF CACHE BOOL "" FORCE)
    set(SDL_OPENGLES OFF CACHE BOOL "" FORCE)
    set(SDL_VULKAN ON CACHE BOOL "" FORCE)
    set(SDL_JOYSTICK ON CACHE BOOL "" FORCE)
    set(SDL_HAPTIC ON CACHE BOOL "" FORCE)
    set(SDL_HIDAPI ON CACHE BOOL "" FORCE)
    set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC_PIC ON CACHE BOOL "" FORCE)
    set(RTY_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS}")
    set(CMAKE_EXE_LINKER_FLAGS "")
    add_subdirectory(3rdparty/SDL2)
    set(CMAKE_EXE_LINKER_FLAGS "${RTY_EXE_LINKER_FLAGS}")
    set(ARCHITECTURE x86_64)
    add_subdirectory(3rdparty/ffmpeg-core)
    if(MINGW)
        target_link_libraries(ffmpeg INTERFACE bcrypt ole32 user32 ws2_32 secur32)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_options(ffmpeg INTERFACE LINKER:--exclude-libs,ALL)
    endif()

    set(FT_DISABLE_ZLIB ON CACHE BOOL "" FORCE)
    set(FT_DISABLE_BZIP2 ON CACHE BOOL "" FORCE)
    set(FT_DISABLE_PNG ON CACHE BOOL "" FORCE)
    set(FT_DISABLE_HARFBUZZ ON CACHE BOOL "" FORCE)
    set(FT_DISABLE_BROTLI ON CACHE BOOL "" FORCE)
    set(SKIP_INSTALL_ALL ON CACHE BOOL "" FORCE)
    add_subdirectory(3rdparty/freetype EXCLUDE_FROM_ALL)
    set_target_properties(freetype PROPERTIES POSITION_INDEPENDENT_CODE ON)

    add_subdirectory(core/shader/recompiler)
    add_subdirectory(core/Decoder/Jpeg)
    add_subdirectory(core/Decoder/Png)
    add_subdirectory(core/libs)
else()
    # Loud: without them the game's runtime libraries (libs/*.prx, libScePad.prx among them) are
    # not built at all, and GTSport.exe fails to load them.
    message(WARNING "3rdparty submodules are missing: the runtime libraries (libs/*.prx) will NOT be built, "
                    "only the Ryty CLI and unit tests. Run 'git submodule update --init --recursive' and "
                    "configure again.")
endif()
