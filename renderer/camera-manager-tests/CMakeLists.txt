cmake_minimum_required(VERSION 3.24)
project(RytyCameraTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
enable_testing()
if(NOT DEFINED SHADGT)
    message(FATAL_ERROR "Pass -DSHADGT=<full checkout with this patch applied>")
endif()
set(WORKSPACE "${CMAKE_CURRENT_BINARY_DIR}")
add_library(camera_settings STATIC "${SHADGT}/ryty_gpu/ryty_camera_settings.cpp")
target_include_directories(camera_settings PUBLIC "${SHADGT}" "${SHADGT}/externals/json/single_include")
add_executable(camera_settings_tests tests/settings_tests.cpp)
target_link_libraries(camera_settings_tests PRIVATE camera_settings)
add_executable(camera_runtime_tests tests/runtime_tests.cpp)
add_executable(camera_validation_tests tests/validation_tests.cpp)
target_link_libraries(camera_validation_tests PRIVATE camera_settings)
add_test(NAME camera_executable_validation_and_hook_rollback COMMAND camera_validation_tests)
add_executable(camera_input_tests tests/input_tests.cpp)
add_executable(camera_sensitivity_tests tests/sensitivity_runtime_tests.cpp)
target_include_directories(camera_sensitivity_tests PRIVATE "${SHADGT}/externals/sdl3/include")
target_link_libraries(camera_sensitivity_tests PRIVATE camera_settings bcrypt)
add_test(NAME camera_sensitivity_actual_runtime COMMAND camera_sensitivity_tests)
target_include_directories(camera_input_tests PRIVATE "${SHADGT}" "${SHADGT}/externals/sdl3/include")
add_test(NAME camera_keyboard_gamepad_event_routing COMMAND camera_input_tests)
target_link_libraries(camera_runtime_tests PRIVATE camera_settings bcrypt)
add_executable(camera_overlay_preview tests/overlay_preview.cpp
    "${SHADGT}/ryty_gpu/ryty_camera_overlay.cpp"
    "${SHADGT}/externals/imgui/imgui.cpp" "${SHADGT}/externals/imgui/imgui_draw.cpp"
    "${SHADGT}/externals/imgui/imgui_widgets.cpp" "${SHADGT}/externals/imgui/imgui_tables.cpp"
    "${SHADGT}/externals/imgui/backends/imgui_impl_dx11.cpp")
target_include_directories(camera_overlay_preview PRIVATE "${SHADGT}/src" "${SHADGT}/externals/imgui")
target_compile_definitions(camera_overlay_preview PRIVATE IMGUI_USER_CONFIG="imgui/imgui_config.h")
target_link_libraries(camera_overlay_preview PRIVATE camera_settings d3d11 dxgi d3dcompiler)
if(MINGW)
    foreach(target camera_settings_tests camera_runtime_tests camera_overlay_preview)
        target_link_options(${target} PRIVATE -static)
    endforeach()
endif()
add_test(NAME preset_persistence_and_isolation COMMAND camera_settings_tests "${CMAKE_CURRENT_BINARY_DIR}/test-settings.json")
add_test(NAME camera_runtime_and_car_identity COMMAND camera_runtime_tests)
add_test(NAME actual_imgui_offscreen_render COMMAND camera_overlay_preview)
add_test(NAME sensitivity_widget_publishes_exact_value COMMAND camera_overlay_preview --size=1280x800 --sensitivity-regression)
set_tests_properties(sensitivity_widget_publishes_exact_value PROPERTIES WORKING_DIRECTORY "${WORKSPACE}" RUN_SERIAL TRUE)
foreach(value -15 0 15)
    add_test(NAME sensitivity_widget_value_${value} COMMAND camera_overlay_preview --size=1280x800 --sensitivity-regression --sensitivity-target=${value})
    set_tests_properties(sensitivity_widget_value_${value} PROPERTIES WORKING_DIRECTORY "${WORKSPACE}" RUN_SERIAL TRUE)
endforeach()
add_test(NAME recover_docked_and_offscreen_camera_window COMMAND camera_overlay_preview --dock-regression)
add_test(NAME overlay_1080p_bounds COMMAND camera_overlay_preview --size=1920x1080)
add_test(NAME overlay_4k_bounds COMMAND camera_overlay_preview --size=3840x2160)
add_test(NAME overlay_retains_relative_layout_after_resolution_change COMMAND camera_overlay_preview --size=3840x2160 --resize-regression)
set_tests_properties(overlay_retains_relative_layout_after_resolution_change PROPERTIES WORKING_DIRECTORY "${WORKSPACE}" RUN_SERIAL TRUE)
set_tests_properties(overlay_1080p_bounds overlay_4k_bounds PROPERTIES WORKING_DIRECTORY "${WORKSPACE}" RUN_SERIAL TRUE)
if(MSVC)
    target_compile_definitions(camera_settings PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
set_tests_properties(preset_persistence_and_isolation camera_runtime_and_car_identity actual_imgui_offscreen_render recover_docked_and_offscreen_camera_window PROPERTIES WORKING_DIRECTORY "${WORKSPACE}")

target_compile_definitions(camera_overlay_preview PRIVATE CAMERA_TEST_FONT="${SHADGT}/src/imgui/renderer/fonts/NotoSans-Regular.ttf")
